\newtcbox{\code}{
    on line,                      % Allows it to sit inside a sentence
    arc=3pt,                      % The "roundness" of the corners
    colback=black!8!white,        % Lighter gray (8% black)
    colframe=white,               % Hides the outer border line
    boxrule=0pt,                  % Removes the border thickness
    boxsep=0pt,                   % Internal spacing
    left=3pt,                     % Left padding
    right=3pt,                    % Right padding
    top=2pt,                      % Top padding
    bottom=2pt,                   % Bottom padding
    fontupper=\ttfamily           % Makes the font Monospaced (Typewriter)
}
