\subsection{System Feature 5: External Calendar Syncing} 


\subsubsection{Description}

After a mentee RSVP's for an event, the event should be automatically synched
with Google Calendar or Microsoft Outlook Calendar.
\textit{(Priority: Medium)}



\subsubsection{Functional Requirements}

\begin{itemize}[label=$\bullet$]
    \item \textbf{ECS-1:} RSVP'd events shall be automatically integrated into
    Google Calendar and Microsoft Outlook Calendar.

    \item \textbf{ECS-2:} Google and Microsoft Outlook Calendars shall contain
    same event title, date, start time, end time, location, and description as
    application.
\end{itemize}



\subsubsection{Use Cases}

% Use Case 9
\begin{usecase}{UC-9: Sync Event to External Calendar} 
    \textbf{Primary Actor}     & Mentee                                          \\ \hline
    \textbf{Secondary Actors:} & Microsoft Graph API and Google Calendar API     \\ \hline
    \textbf{Description}       & After a mentee RSVPs to an event, the
                                 application will automatically push event
                                 details to their Google Calendar or Microsoft
                                 Outlook Calendars.                              \\ \hline
    \textbf{Preconditions}     & PRE-1: Mentee is RSVP'd to an event.            \newline
                                 PRE-2: Mentee has linked their Google or
                                 Microsoft Outlook calendar(s).                  \\ \hline
    \textbf{Postconditions}    & POST-1: Event is successfully integrated into
                                 external calendar.                              \\ \hline
    \textbf{Normal Flow}       & 
    \begin{tabenum}
        \item Mentee RSVP's to event.
        \item Application sends event details to external calendar APIs.
        \item Event is created on external calendar.
    \end{tabenum} \\
\end{usecase}



\subsubsection{User Stories}

\begin{itemize}[leftmargin=*, label=$\square$]
    \item \textbf{As a mentee}, I want my RSVP'd events to automatically sync to
    my Google/Microsoft Calendars so that I don't have to manually add the
    event.
    \begin{itemize}[label=$\circ$, nosep]
        \item \textit{Acceptance:} After a mentee RSVPs, the event is displayed
        on their external calendar.
        
        \item \textit{Acceptance:} Events on application and external calendar
        share the same event title, date, start time, end time, location, and
        description.
    \end{itemize}
\end{itemize}
