\subsection{Security}

\begin{itemize}
    \item \textbf{SEC-1:} There shall be no exposure of confidential information
    (PII).
    
    \item \textbf{SEC-2:} Mentors shall only have access to their mentor
    dashboard.
    
    \item \textbf{SEC-3:} Mentees shall only have access to their personalized
    calendar.
    
    \item \textbf{SEC-4:} Super Admin is only user who can promote and demote
    other users.
    
    \item \textbf{SEC-5:} User authentication shall be handled by OU SSO and
    Google Login/OAuth.
\end{itemize}
