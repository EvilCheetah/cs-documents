\item[\textbf{Problem 1}]
Compute the following with \textsc{Detailed Mathematical Steps}. (Hints: Use
Fermat Theorem, Euler Theorem, properties of totient functions, etc)
\begin{enumerate}[label=(\arabic*), leftmargin=*]
    \item $1234^{16} \bmod 17$
    \item $54^{51} \bmod 17$
    \item $\phi(51)$
    \item $\gcd(33, 121)$
    \item $2^{-1} \bmod 17$ (i.e., multiplicative inverse of $2 \bmod 17$)
    \item $\text{ind}_{2,5}(4)$
    \item $\phi(8000)$
    \item $\phi(\num{98803519})$
    \item $\phi(\num{999866001989})$ \textcolor{red}{for the graduate session
    only.}
\end{enumerate}

\Solution

\begin{enumerate}[label=(\arabic*), leftmargin=*]
    \item $1234^{16} \bmod 17$
    \begin{solution}
        By Fermat's Little Theorem, since $17$ is prime, we know that:
        \begin{alignat*}{2}
            a^{p-1}     &\equiv 1 \pmod p   \\
            1234^{16}   &\equiv 1 \pmod{17} \\
            1234^{17-1} &\equiv 1 \pmod{17} \\
        \end{alignat*}
        Therefore, the answer is $1$.
    \end{solution}


    \item $54^{51} \bmod 17$
    \begin{solution}
        Since $17$ is prime, by Fermat's Little Theorem, $\phi(17) = 16$.
        Applying modular exponentiation and then the modular multiplication
        rule:
        \begin{alignat*}{2}
            54^{51} \bmod 17 &= 54^{51 \bmod 16} \bmod 17  \\
                             &= 54^3 \bmod 17              \\
                             &= (54 \bmod 17)^3 \bmod 17   \\
                             &= 3^3 \bmod 17               \\
                             &= 27 \bmod 17                \\
                             &= 10
        \end{alignat*}
        Therefore, the answer $10$.
    \end{solution}
    
    
    \item $\phi(51)$
    \begin{solution}
        By Euler's Totient Function, we know that:
        \begin{alignat*}{2}
            \phi(51)    &= \phi(17 \times 3)       \\
                        &= \phi(17) \times \phi(3) \\
                        &= 16 \times 2             \\
                        &= 32                      \\
        \end{alignat*}
        Therefore, the answer is $32$.
    \end{solution}

    
    \item $\gcd(33, 121)$
    \begin{solution}
        By finding the prime factorization of both numbers, we have:
        \begin{align*}
            33  &= 3 \cdot 11 \\
            121 &= 11^2
        \end{align*}
        The greatest prime factor shared by both numbers is $11$. Therefore, 
        the asnwer to $\gcd(33, 121)$ is $11$.
    \end{solution}

    
    \item $2^{-1} \bmod 17$ (i.e., multiplicative inverse of $2 \bmod 17$)
    \begin{solution}
        Since $17$ is prime and $2$ is not divisible by $17$, by Fermat's Little
        Theorem, $2^{17-1} \equiv 1 \pmod{17}$. This implies that $2 \cdot
        2^{15} \equiv 1 \pmod{17}$, making the multiplicative inverse $2^{15}
        \bmod 17$. Applying modular exponentiation rules:
        \begin{alignat*}{2}
            2^{-1} \bmod 17 &= 2^{15} \bmod 17 \\
                            &= (2^4)^3 \cdot 2^3 \bmod 17 \\
                            &= 16^3 \cdot 8 \bmod 17 \\
                            &= (-1)^3 \cdot 8 \bmod 17 \\
                            &= -8 \bmod 17 \\
                            &= 9
        \end{alignat*}
        Therefore, the answer is $9$.
    \end{solution}
    
    
    \item $\text{ind}_{2,5}(4)$
    \begin{solution}
        By using properties of discrete logarithms, we can get the following:
        \begin{align*}
            \text{ind}_{2,5}(4)  &= \text{ind}_{2,5}(2^2)                        \\
                                 &= (2 \times \text{ind}_{2,5}(2)) \bmod \phi(5) \\
                                 &= (2 \times \text{ind}_{2,5}(2)) \bmod 4       \\
                                 &= (2 \times 1) \bmod 4                         \\
                                 &= 2 \bmod 4                                    \\
                                 &= 2
        \end{align*}
        Therefore, the answer is $2$.
    \end{solution}

    
    \item $\phi(8000)$
    \begin{solution}
        By Euler's Totient Function, we know that:
        \begin{alignat*}{2}
            \phi(8000)  &= \phi(8 \times 1000)                             \\
                        &= \phi(2^3 \times 10^3)                           \\
                        &= \phi(2^3 \times (5 \cdot 2)^3)                  \\
                        &= \phi(2^3 \times 5^3 \times 2^3)                 \\
                        &= \phi(2^{3+3} \times 5^3)                        \\
                        &= \phi(2^6 \times 5^3)                            \\
                        &= \phi(2^6) \times \phi(5^3)                      \\
                        &= ((2 - 1) \cdot 2^5) \times ((5 - 1) \cdot 5^2)  \\
                        &= (1 \cdot 32) \times (4 \cdot 25)                \\
                        &= 32 \times 100                                   \\
                        &= 3200                                            \\
        \end{alignat*}
        Therefore, the answer is $3200$.
    \end{solution}

    
    \item $\phi(\num{98803519})$
    \begin{solution}
        First, let's find the prime factorization of $\num{98803519}$:
        \begin{alignat*}{2}
            \num{98803519}   &= \num{9949} \times \num{9931}
        \end{alignat*}

        Both $\num{9949}$ and $\num{9931}$, hence we can use Euler's Totient Function
        properties for distinct primes:
        \begin{alignat*}{2}
            \phi(\num{98803519}) &= \phi(\num{9949} \times \num{9931})       \\
                                 &= \phi(\num{9949}) \times \phi(\num{9931}) \\
                                 &= \num{9948} \times \num{9930}             \\
                                 &= \num{98783640}
        \end{alignat*}
        Therefore, the answer is $\num{98783640}$.
    \end{solution}

    
    \item $\phi(\num{999866001989})$ \textcolor{red}{for the graduate session only.}
    \begin{solution}
        First, let's find the prime factorization of $\num{999866001989}$:
        \begin{alignat*}{2}
            \num{999866001989}   &= \num{999883} \times \num{999983}
        \end{alignat*}

        Both $\num{999883}$ and $\num{999983}$, hence we can use Euler's Totient Function
        properties for distinct primes:
        \begin{alignat*}{2}
            \phi(\num{98803519}) &= \phi(\num{999883} \times \num{999983})       \\
                                 &= \phi(\num{999883}) \times \phi(\num{999983}) \\
                                 &= \num{999882} \times \num{999982}             \\
                                 &= \num{999864002124}
        \end{alignat*}
        Therefore, the answer is $\num{999864002124}$.
    \end{solution}
\end{enumerate}
