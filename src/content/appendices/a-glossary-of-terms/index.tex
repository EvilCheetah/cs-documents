\section{Glossary of Terms}

\begin{enumerate}
    \item \textbf{Super Admin} --- The leader of the student organization who
    can promote and demote members

    \item \textbf{Mentor} --- Upperclassmen responsible for creating and
    organizing events

    \item \textbf{Mentee} --- Lower-classmen who will be invited to events
    
    \item \textbf{Cohort} --- A group of mentees in the same academic year, such
    as Freshman, Sophomore, Junior, Senior
\end{enumerate}
