% --- C++ Color Definitions ---
\definecolor{codebackground}{rgb}{0.95, 0.95, 0.96}
\definecolor{codecomment}{rgb}{0.25, 0.50, 0.35}
\definecolor{codekeyword}{rgb}{0.00, 0.30, 0.75}
\definecolor{codestring}{rgb}{0.70, 0.10, 0.10}
\definecolor{codenumber}{rgb}{0.50, 0.50, 0.50}

% --- C++ Style Definition ---
\lstdefinestyle{moderncpp}{
    language=C++,
    backgroundcolor=\color{codebackground},   
    commentstyle=\color{codecomment}\itshape,
    keywordstyle=\color{codekeyword}\bfseries,
    stringstyle=\color{codestring},
    numberstyle=\tiny\color{codenumber},
    basicstyle=\ttfamily\footnotesize,
    breakatwhitespace=false,         
    breaklines=true,                 
    captionpos=b,                    
    keepspaces=true,                 
    numbers=left,                    
    numbersep=10pt,                  
    showspaces=false,                
    showstringspaces=false,
    showtabs=false,                  
    tabsize=4,
    frame=single,
    rulecolor=\color{black!20}
}


% --- Python Color Definitions ---
\definecolor{pybackground}{rgb}{0.97, 0.97, 0.98}
\definecolor{pycomment}{rgb}{0.40, 0.40, 0.40} % Grey for comments
\definecolor{pykeyword}{rgb}{0.00, 0.44, 0.75} % Deep blue for keywords
\definecolor{pystring}{rgb}{0.13, 0.55, 0.13}  % Forest green for strings
\definecolor{pynumber}{rgb}{0.50, 0.00, 0.50}  % Purple for numbers
\definecolor{pybuiltin}{rgb}{0.56, 0.27, 0.52} % Plum for built-ins (like print, len)

% --- Python Style Definition ---
\lstdefinestyle{modernpython}{
    language=Python,
    backgroundcolor=\color{pybackground},
    commentstyle=\color{pycomment}\itshape,
    keywordstyle=\color{pykeyword}\bfseries,
    stringstyle=\color{pystring},
    numberstyle=\tiny\color{pynumber},
    basicstyle=\ttfamily\footnotesize,
    % Customizing to highlight Python built-ins separately
    emph={print,len,range,enumerate,zip,help,input,list,dict,set,int,str,float},
    emphstyle=\color{pybuiltin},
    breakatwhitespace=false,
    breaklines=true,
    captionpos=b,
    keepspaces=true,
    numbers=left,
    numbersep=10pt,
    showspaces=false,
    showstringspaces=false,
    showtabs=false,
    tabsize=4,
    frame=single,
    rulecolor=\color{black!15},
    % Handling Python's specific characters
    upquote=true, 
    extendedchars=true
}


\lstset{style=moderncpp}
\lstset{style=modernpython}
